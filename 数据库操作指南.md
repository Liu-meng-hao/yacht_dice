你的项目已经有以下数据库模型：

### User 模型（用户表）

```python
class User(Base):
    __tablename__ = "users"
    
    id              # 主键，自增
    username        # 用户名，唯一
    nickname        # 昵称
    email           # 邮箱，唯一
    hashed_password # 加密后的密码
    avatar          # 头像
    is_active       # 是否激活
    created_at      # 创建时间
    updated_at      # 更新时间
```

---

### GameRecord 模型（游戏记录表）

```python
class GameRecord(Base):

    scores          # 得分（JSON）
    winner          # 赢家
    status          # 状态
    created_at      # 创建时间
```

## 三、CRUD 操作详解

CRUD = Create（创建）、Read（读取）、Update（更新）、Delete（删除）

---

### 1. Create（创建数据）

#### 示例：创建一个新用户

```python
from app.db.session import SessionLocal
from app.models.user import User

# 获取数据库会话
db = SessionLocal()

try:
    # 创建用户对象
    new_user = User(
        username="test_user",
        nickname="测试用户",
        email="test@example.com",
        hashed_password="hashed_password_here"
    )
    
    # 添加到会话
    db.add(new_user)
    
    # 提交到数据库
    db.commit()
    
    # 刷新对象（获取自动生成的 id）
    db.refresh(new_user)
    
    print(f"用户创建成功！ID: {new_user.id}")
    
finally:
    # 关闭会话
    db.close()
```

---

### 2. Read（读取数据）

#### 示例 1：根据 ID 查询用户

```python
from app.db.session import SessionLocal
from app.models.user import User

db = SessionLocal()

try:
    # 查询 ID 为 1 的用户
    user = db.query(User).filter(User.id == 1).first()
    
    if user:
        print(f"用户名: {user.username}")
        print(f"昵称: {user.nickname}")
    else:
        print("用户不存在")
        
finally:
    db.close()
```

---

#### 示例 2：查询所有用户

```python
from app.db.session import SessionLocal
from app.models.user import User

db = SessionLocal()

try:
    # 查询所有用户
    users = db.query(User).all()
    
    print(f"共找到 {len(users)} 个用户:")
    for user in users:
        print(f"  - {user.username}")
        
finally:
    db.close()
```

---

#### 示例 3：条件查询

```python
from app.db.session import SessionLocal
from app.models.user import User

db = SessionLocal()

try:
    # 查询活跃用户
    active_users = db.query(User).filter(User.is_active == True).all()
    
    # 根据用户名查询
    user = db.query(User).filter(User.username == "test_user").first()
    
    # 多条件查询
    users = db.query(User).filter(
        User.is_active == True,
        User.username.like("%test%")
    ).all()
        
finally:
    db.close()
```

---

### 3. Update（更新数据）

#### 示例：更新用户信息

```python
from app.db.session import SessionLocal
from app.models.user import User

db = SessionLocal()

try:
    # 先查询用户
    user = db.query(User).filter(User.id == 1).first()
    
    if user:
        # 更新字段
        user.nickname = "新昵称"
        user.email = "new_email@example.com"
        
        # 提交
        db.commit()
        
        print("用户信息已更新")
    else:
        print("用户不存在")
        
finally:
    db.close()
```

---

### 4. Delete（删除数据）

#### 示例：删除用户

```python
from app.db.session import SessionLocal
from app.models.user import User

db = SessionLocal()

try:
    # 先查询用户
    user = db.query(User).filter(User.id == 1).first()
    
    if user:
        # 删除
        db.delete(user)
        db.commit()
        print("用户已删除")
    else:
        print("用户不存在")
        
finally:
    db.close()
```

## 四、在 API 中使用数据库

### 使用依赖注入（推荐方式）

FastAPI 推荐使用依赖注入来获取数据库会话。

#### 示例：在 API 中创建游戏记录

```python
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from app.db.session import get_db
from app.models.game import GameRecord
from app.schemas.game import GameMode
from app.core.response import ApiResponse
import json

router = APIRouter()


@router.post("/game/records")
async def save_game_record(
    game_id: str,
    game_mode: GameMode,
    player_names: list,
    scores: dict,
    winner: str,
    db: Session = Depends(get_db)
):
    """保存游戏记录到数据库"""
    
    # 创建游戏记录对象
    record = GameRecord(
        game_id=game_id,
        game_mode=game_mode.value,
        player_names=json.dumps(player_names),
        scores=json.dumps(scores),
        winner=winner,
        status="finished"
    )
    
    # 保存到数据库
    db.add(record)
    db.commit()
    db.refresh(record)
    
    return ApiResponse.success(
        data={"record_id": record.id},
        msg="游戏记录已保存"
    )


@router.get("/game/records")
async def get_game_records(db: Session = Depends(get_db)):
    """获取游戏记录列表"""
    
    records = db.query(GameRecord).order_by(GameRecord.created_at.desc()).all()
    
    return ApiResponse.success(
        data={"records": records},
        msg="获取成功"
    )
```

## 📝 完整示例：保存游戏结束记录

当游戏结束时，自动保存到数据库：

```python
from app.db.session import SessionLocal
from app.models.game import GameRecord
from app.game.game_manager import GameManager
import json


def save_game_to_db(game_id: str):
    """游戏结束后保存到数据库"""
    
    game = GameManager.get_game(game_id)
    if not game or game.status != "finished":
        return
    
    db = SessionLocal()
    try:
        # 获取赢家
        winner = game.get_winner()
        
        # 准备玩家得分
        player_scores = {}
        for player in game.players:
            player_scores[player.player_id] = {
                "name": player.name,
                "scores": player.scores,
                "total_score": player.total_score
            }
        
        # 创建记录
        record = GameRecord(
            game_id=game.game_id,
            game_mode=game.game_mode,
            player_names=json.dumps([p.name for p in game.players]),
            scores=json.dumps(player_scores),
            winner=winner.player_id if winner else None,
            status="finished"
        )
        
        db.add(record)
        db.commit()
        
        print(f"游戏记录已保存，ID: {record.id}")
        
    finally:
        db.close()
```

## 🎯 总结

| 操作 | 方法 |
|------|------|
| **创建表** | `python scripts/init_db.py` |
| **创建数据** | `db.add()` + `db.commit()` |
| **查询数据** | `db.query().filter().first()` / `.all()` |
| **更新数据** | 修改字段 + `db.commit()` |
| **删除数据** | `db.delete()` + `db.commit()` |
| **在 API 中使用** | `Depends(get_db)` |

---

## ⚠️ 注意事项

1. **记得关闭会话**：使用完 `db` 后一定要 `db.close()`
2. **异常处理**：数据库操作要加 try-finally
3. **事务安全**：多个操作要在一个事务中完成
